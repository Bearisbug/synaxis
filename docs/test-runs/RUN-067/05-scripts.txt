# dropline <file> <key>: deletes the one line of that key
dropline() { $PY - "$@" <<'EOF'
import sys
path, key = sys.argv[1], sys.argv[2]
lines = open(path, encoding='utf-8').read().split('\n')
hit = [i for i, l in enumerate(lines) if l.lstrip().startswith('"' + key + '":')]
if len(hit) != 1:
    sys.exit(f"{path}: {len(hit)} lines of {key}")
del lines[hit[0]]
open(path, 'w', encoding='utf-8').write('\n'.join(lines))
print(f"deleted the {key} line of {path}")
EOF
}

===== count.py
"""Counts JUnit XML results. Usage:
  count.py summary <results-dir> <since-epoch> [prefix-regex ...]
  count.py gate <xml-file> <since-epoch>
Files older than <since-epoch> are ignored (a stale XML from an earlier run is not counted)."""
import os
import re
import sys
import xml.etree.ElementTree as ET


def fresh(path, since):
    return os.path.getmtime(path) >= since


def summary(d, since, prefixes):
    rows = []
    for name in sorted(os.listdir(d)):
        if not name.endswith('.xml'):
            continue
        p = os.path.join(d, name)
        if not fresh(p, since):
            continue
        r = ET.parse(p).getroot()
        rows.append((r.get('name'), int(r.get('tests')), int(r.get('failures')), int(r.get('errors')),
                     int(r.get('skipped'))))
    stale = [n for n in os.listdir(d) if n.endswith('.xml') and not fresh(os.path.join(d, n), since)]
    t = sum(x[1] for x in rows)
    f = sum(x[2] for x in rows)
    e = sum(x[3] for x in rows)
    s = sum(x[4] for x in rows)
    print(f"{d}: classes={len(rows)} tests={t} failures={f} errors={e} skipped={s} (stale files ignored: {len(stale)})")
    for pre in prefixes:
        sub = [x for x in rows if re.match(pre, x[0])]
        print(f"== {pre}: {len(sub)} classes, {sum(x[1] for x in sub)} tests, "
              f"{sum(x[2] + x[3] for x in sub)} failed, {sum(x[4] for x in sub)} skipped")
    sk = [x for x in rows if x[4]]
    if sk:
        print("classes with skipped tests:")
        for x in sk:
            print(f"  {x[0].replace('io.github.bearisbug.synaxis.app.', '')} skipped={x[4]}")
    bad = [x for x in rows if x[2] or x[3]]
    for x in bad:
        print(f"  FAILED {x[0]} failures={x[2]} errors={x[3]}")


def gate(p, since):
    if not os.path.exists(p) or not fresh(p, since):
        print(f"{os.path.basename(p)}: no fresh result")
        return
    r = ET.parse(p).getroot()
    print(f"{os.path.basename(p)}: tests={r.get('tests')} failures={r.get('failures')} errors={r.get('errors')} "
          f"skipped={r.get('skipped')}")
    for tc in r.iter('testcase'):
        for fl in list(tc.iter('failure')) + list(tc.iter('error')):
            text = fl.text or ''
            red = re.findall(r'\((\w+\.java:\d+)\)', text)
            own = [x for x in red if 'GateTest' in x]
            print(f"  red: {tc.get('name')} at {own[0] if own else (red[0] if red else '?')}")
            for line in (fl.get('message') or '').splitlines():
                if line.strip():
                    print(f"    {line}")
    so = r.find('system-out')
    if so is not None and so.text:
        for line in so.text.splitlines():
            if 'gate:' in line:
                print(f"  system-out: {line.strip()}")


if __name__ == '__main__':
    if sys.argv[1] == 'summary':
        summary(sys.argv[2], float(sys.argv[3]), sys.argv[4:])
    else:
        gate(sys.argv[2], float(sys.argv[3]))
