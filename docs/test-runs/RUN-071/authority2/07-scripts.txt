# gt <label> <task> [extra args...]: runs one gradle test task in the worktree, output to $OUT/<label>.txt
gt() {
  local label=$1 task=$2; shift 2
  (cd $WT && ./gradlew $task --rerun "$@" --max-workers=2 --no-daemon --init-script gradle/init/aliyun-mirror.init.gradle.kts > $OUT/$label.txt 2>&1)
  local rc=$?
  echo "$label exit=$rc $(grep -E 'tests completed|BUILD (SUCCESSFUL|FAILED)' $OUT/$label.txt | tr '\n' ' ')"
}
# keep <label> <module-results-dir> <class...>: copies the JUnit XML of the classes
keep() {
  local label=$1 dir=$2; shift 2
  mkdir -p $OUT/xml/$label
  for c in "$@"; do cp $WT/$dir/TEST-$c.xml $OUT/xml/$label/ 2>/dev/null; done
}
PY=/Users/bug/Documents/Projects/Minecraft/synaxis/.venv/bin/python
# proof <label> <file> <old> <new> <task> <class> <results-dir> [extra gradle args]: edit, run, keep XML, revert, check clean
proof() {
  local label=$1 file=$2 old=$3 new=$4 task=$5 cls=$6 dir=$7; shift 7
  $PY $OUT/edit.py $WT/$file "$old" "$new" || { echo "$label EDIT FAILED"; return 1; }
  gt $label $task --tests $cls "$@"
  keep $label $dir $cls
  (cd $WT && git checkout HEAD -- $file && git status --short --untracked-files=no | grep -q . && echo "$label NOT CLEAN")
}
# lproof <label> <file> <line> <old> <new> <task> <results-dir> <class...> -- [extra gradle args]: line-based edit, run, keep, revert
lproof() {
  local label=$1 file=$2 line=$3 old=$4 new=$5 task=$6 dir=$7; shift 7
  local classes=() extra=()
  while [[ $# -gt 0 && $1 != -- ]]; do classes+=($1); shift; done
  [[ $1 == -- ]] && shift
  extra=("$@")
  $PY $OUT/editline.py $WT/$file $line "$old" "$new" || { echo "$label EDIT FAILED"; return 1; }
  local t=(); for c in $classes; do t+=(--tests $c); done
  gt $label $task $t $extra
  keep $label $dir $classes
  (cd $WT && git checkout HEAD -- $file && git status --short --untracked-files=no | grep -q . && echo "$label NOT CLEAN")
}

import sys
path, old, new = sys.argv[1], sys.argv[2], sys.argv[3]
s = open(path).read()
n = s.count(old)
if n != 1:
    sys.exit(f"{path}: {n} occurrences of {old!r}")
open(path, 'w').write(s.replace(old, new))

===== editline.py
import sys
path, line, old, new = sys.argv[1], int(sys.argv[2]), sys.argv[3], sys.argv[4]
lines = open(path).read().split('\n')
if lines[line - 1].count(old) != 1:
    sys.exit(f"{path}:{line}: {lines[line - 1]!r} does not hold {old!r} once")
lines[line - 1] = lines[line - 1].replace(old, new)
open(path, 'w').write('\n'.join(lines))

===== summarize.py
import sys, glob, os, re, xml.etree.ElementTree as ET
for d in sys.argv[1:]:
    for f in sorted(glob.glob(f"xml/{d}/*.xml")):
        r = ET.parse(f).getroot()
        print(f"{d}: {r.get('name').split('.')[-1]} tests={r.get('tests')} failures={r.get('failures')} errors={r.get('errors')} skipped={r.get('skipped')}")
        for tc in r.iter('testcase'):
            for fl in list(tc.findall('failure')) + list(tc.findall('error')):
                msg = (fl.get('message') or '').replace('\n', ' ')
                if len(msg) > 330: msg = msg[:150] + ' … ' + msg[-170:]
                lines = re.findall(r'\((\w+Test\.java:\d+)\)', fl.text or '')
                print(f"   {tc.get('name')} @ {lines[:1]} | {msg}")
