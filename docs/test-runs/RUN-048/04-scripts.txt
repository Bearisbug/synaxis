#!/bin/zsh
# usage: g.sh <label> [nodb] -- <gradle args...>   one Gradle in the m3e-follow worktree, one at a time
label=$1; shift
nodb=0
if [ "$1" = nodb ]; then nodb=1; shift; fi
[ "$1" = "--" ] && shift
source /tmp/m3e-run048/env.sh
if (( nodb )); then unset SYNAXIS_TEST_PG_URL SYNAXIS_TEST_PG_USER SYNAXIS_TEST_PG_PASSWORD SYNAXIS_TEST_REDIS_URL; fi
cd /Users/bug/Documents/Projects/Minecraft/synaxis-wt/m3e-follow
while pgrep -f 'GradleWrapperMain|GradleDaemon|GradleMain' >/dev/null; do echo "waiting for another Gradle: $(pgrep -fl 'GradleWrapperMain|GradleDaemon|GradleMain' | cut -c1-80 | tr '\n' ';')"; sleep 20; done
log=/tmp/m3e-run048/logs/$label.log
start=$(date +%s)
echo "# $label $(date '+%F %T') HEAD $(git rev-parse --short=10 HEAD) dirty=$(git status --porcelain | wc -l | tr -d ' ') load1 $(sysctl -n vm.loadavg | awk '{print $2}') swap_free $(sysctl -n vm.swapusage | sed -E 's/.*free = ([0-9.]+M).*/\1/') db=$((1-nodb))" > $log
echo "# ./gradlew --no-daemon --init-script gradle/init/aliyun-mirror.init.gradle.kts --max-workers=4 $*" >> $log
./gradlew --no-daemon --init-script gradle/init/aliyun-mirror.init.gradle.kts --max-workers=4 "$@" >> $log 2>&1
rc=$?
echo "# rc=$rc wall=$(( $(date +%s) - start ))s end $(date '+%T')" >> $log
tail -1 $log; grep -E 'BUILD (SUCCESSFUL|FAILED)|tests completed' $log | head -3
exit $rc

=== sum.sh
#!/bin/zsh
# usage: sum.sh <label>   totals of the JUnit XML of every module after one gate, written to evidence/<label>.txt
setopt nullglob
label=$1
cd /Users/bug/Documents/Projects/Minecraft/synaxis-wt/m3e-follow
out=/tmp/m3e-run048/evidence/$label.txt
{
  head -2 /tmp/m3e-run048/logs/$label.log
  grep -E "BUILD (SUCCESSFUL|FAILED)|tests completed|# rc=" /tmp/m3e-run048/logs/$label.log
  T=0; S=0; F=0; E=0
  for d in build */build; do
    files=($d/test-results/test/TEST-*.xml)
    (( ${#files} )) || continue
    t=0; s=0; f=0; e=0
    for x in $files; do
      h=$(grep -o '<testsuite [^>]*' $x | head -1)
      t=$((t + $(echo $h | sed -E 's/.* tests="([0-9]+)".*/\1/')))
      s=$((s + $(echo $h | sed -E 's/.* skipped="([0-9]+)".*/\1/')))
      f=$((f + $(echo $h | sed -E 's/.* failures="([0-9]+)".*/\1/')))
      e=$((e + $(echo $h | sed -E 's/.* errors="([0-9]+)".*/\1/')))
    done
    echo "module ${d%/build}: tests=$t skipped=$s failures=$f errors=$e"
    T=$((T+t)); S=$((S+s)); F=$((F+f)); E=$((E+e))
  done
  echo "total: tests=$T skipped=$S failures=$F errors=$E"
  for c in app.shard.ShardGatesTest app.store.TicketIssuerTickThreadIntegrationTest app.store.TransferTicketsIntegrationTest app.TransferAdmissionIntegrationTest app.shard.ShardServiceImplTest; do
    x=server-app/build/test-results/test/TEST-io.github.bearisbug.synaxis.$c.xml
    echo "$c: $(grep -o '<testsuite [^>]*' $x | grep -o 'tests="[0-9]*" skipped="[0-9]*" failures="[0-9]*" errors="[0-9]*"') ownership_exceptions=$(grep -c AcquirableOwnershipException $x)"
  done
  echo "server-app classes with skipped>0: $(grep -l 'skipped="[1-9]' server-app/build/test-results/test/TEST-*.xml 2>/dev/null | wc -l | tr -d ' ')"
  echo "AcquirableOwnershipException in any XML: $(grep -l AcquirableOwnershipException */build/test-results/test/TEST-*.xml build/test-results/test/TEST-*.xml 2>/dev/null | wc -l | tr -d ' ') files"
} > $out
cat $out

#!/bin/zsh
# the counterproofs of RUN-048, one Gradle at a time; each edit is undone with git checkout and checked with git status

OUT=/tmp/m3e-run048/evidence/03-cases.txt
P=io.github.bearisbug.synaxis.app
report() { # label classes...
  local label=$1; shift
  echo "---- $label" >> $OUT
  grep -E "tests completed|BUILD (SUCCESSFUL|FAILED)|# rc=|错误|error:" /tmp/m3e-run048/logs/$label.log | head -8 >> $OUT
  for c in "$@"; do
    x=server-app/build/test-results/test/TEST-$P.$c.xml
    echo "$c $(grep -o '<testsuite [^>]*' $x | grep -o 'tests="[0-9]*" skipped="[0-9]*" failures="[0-9]*" errors="[0-9]*"')" >> $OUT
    grep -A4 -E '<failure|<error' $x | grep -E "(failure|error) message|at $P" | sed -E 's/ type="[^"]*">.*//' | sed 's/&lt;/</g; s/&gt;/>/g; s/&quot;/"/g; s/&#10;/ /g' | cut -c1-420 >> $OUT
  done
}
run() { # label nodb|db classes...
  local label=$1 mode=$2; shift 2
  local args=()
  for c in "$@"; do args+=(--tests "$P.$c"); done
  if [ $mode = nodb ]; then /tmp/m3e-run048/g.sh $label nodb -- :server-app:test $args >/dev/null
  else /tmp/m3e-run048/g.sh $label -- :server-app:test $args >/dev/null; fi
  report $label "$@"
}
edit() { # file perl-substitution
  perl -0pi -e "$2" $1
  echo "edit $1:" >> $OUT; git diff -U0 $1 | grep -E '^[-+][^-+]' >> $OUT
}
restore() { git checkout -- "$@"; echo "restored: git status --short -> [$(git status --short | tr '\n' ' ')]" >> $OUT; }
