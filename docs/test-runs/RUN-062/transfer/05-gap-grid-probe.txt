# GAP-grid (read-only check asked of this group): can a player duplicate items with a pickup or a drop across a grid cell border
# between the transfer freeze and the save? Answer from the code: no (docs/DESIGN-PROD.md §28.2, the risk row on cross-cell pickups).
# The check turned up one thread-ownership defect, measured here with a probe test that is not committed: the picker's tick thread
# removes an item entity that another tick thread owns (LivingEntity.java:242). The probe was copied to
# src/test/java/io/github/bearisbug/synaxis/entity/CrossCellPickupProbeTest.java, run with
# ./gradlew :test --rerun --tests io.github.bearisbug.synaxis.entity.CrossCellPickupProbeTest --max-workers=4 --no-daemon
# --init-script gradle/init/aliyun-mirror.init.gradle.kts (STRICT adds the three strict grid properties), then deleted; each batch held
# the heavy lock. HEAD 703528c9e9 (no engine source differs from main 23982267e9). The probe builds its own 4-thread grid of 2 by 2
# chunk cells, so both modes tick the picker on Syn-Tick-0 and the item on Syn-Tick-3; the test JVM runs with assertions on, so
# assertOwnership checks in the default mode as well. Production runs without -ea and without synaxis.acquirable-strict: there the
# removal goes ahead on the wrong thread, which this probe does not measure.

  package io.github.bearisbug.synaxis.entity;
  
  import io.github.bearisbug.synaxis.ServerConfig;
  import io.github.bearisbug.synaxis.coordinate.Pos;

  import org.junit.jupiter.api.Test;
  
  import java.util.List;
  import java.util.concurrent.CopyOnWriteArrayList;
  
  import static org.junit.jupiter.api.Assertions.assertEquals;
  import static org.junit.jupiter.api.Assertions.assertNull;
  import static org.junit.jupiter.api.Assertions.assertTrue;
  
  /** Probe, not committed: a picker on one tick thread picks up an item that the next cell's tick thread owns. */
  public class CrossCellPickupProbeTest {
  
      private Env env;
  
      @AfterEach
      void tearDown() {
          env.process().stop();
      }
  
      @Test
      public void anItemAcrossACellBorderIsPickedUpOnce() {
          env = Env.create(settings -> settings.dispatcherThreads(4)
                  .threadProvider(ServerConfig.ThreadProviderKind.GRID)
                  .threadGridExponent(1));
          final Instance instance = env.createFlatInstance();
          // Cells of 2 by 2 chunks: x = 32 is the border of cell (0, 0), id 0, and cell (1, 0), id 3
          final LivingEntity picker = new LivingEntity(EntityType.ZOMBIE);
          picker.setCanPickupItem(true);
          picker.setInstance(instance, new Pos(31.7, 40, 8.5)).join();
          final ItemEntity item = new ItemEntity(ItemStack.of(Material.DIAMOND, 1));
          item.setInstance(instance, new Pos(32.3, 40, 8.5)).join();
          final List<String> pickups = new CopyOnWriteArrayList<>();
          env.process().eventHandler().addListener(PickupItemEvent.class, event -> pickups.add(Thread.currentThread().getName()
                  + " picker-owned=" + event.getLivingEntity().acquirable().isOwned()
                  + " item-owned=" + event.getItemEntity().acquirable().isOwned()));
  
          Throwable failure = null;
          int ticks = 0;
          while (ticks < 20 && failure == null) {
              ticks++;
              try {
                  env.tick();
              } catch (AssertionError e) {
                  failure = e.getCause() == null ? e : e.getCause();
              }
          }
          int laterFailures = 0;
          for (int i = 0; i < 20; i++) {
              try {
                  env.tick();
              } catch (AssertionError _) {
                  laterFailures++;
              }
          }
          System.out.println("PROBE ticks-until-failure=" + ticks + " failure=" + failure);
          if (failure != null) for (StackTraceElement frame : failure.getStackTrace()) System.out.println("PROBE   at " + frame);
          System.out.println("PROBE pickups=" + pickups + " later-failures=" + laterFailures);
          System.out.println("PROBE item removed=" + item.isRemoved() + " pickable=" + item.isPickable()
                  + " claimable-again=" + item.claim());
          assertNull(failure, "the pickup across the border threw on a tick thread");
          assertEquals(1, pickups.size());
          assertTrue(item.isRemoved());
      }
  }

# Reading: on the first tick the picker on Syn-Tick-0 claims the item, the PickupItemEvent fires once (item-owned=false), and
# Entity#remove throws AcquirableOwnershipException at its assertOwnership (Entity.java:1915). The item stays in the world, still
# claimed (claimable-again=false), so nothing picks it up again in the next 20 ticks and no second event fires: no duplication,
# an item left behind. The fix is not this group's; the §28.2 row hands it to the integrate round.
