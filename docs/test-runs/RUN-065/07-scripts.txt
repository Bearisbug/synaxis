=== holder.zsh
#!/bin/zsh
# RUN-065: takes /tmp/synaxis-heavy.lock once and runs the jobs of queue/ in name order while holding it; more jobs may be
# dropped into queue/ meanwhile. Releases on the file release, after 10 min with nothing queued, or on any exit (top-level trap).
T=/tmp/followup-065.KjAtEm
LOCK=/tmp/synaxis-heavy.lock
TL=$T/lock.log
echo "$(date '+%F %T') holder pid=$$ waiting for $LOCK" >> $TL
until mkdir $LOCK 2>/dev/null; do
  echo "$(date '+%F %T') busy: $(cat $LOCK/owner 2>/dev/null)" >> $TL
  sleep 60
done
trap 'echo "$(date "+%F %T") released by pid $$" >> /tmp/followup-065.KjAtEm/lock.log; rm -rf /tmp/synaxis-heavy.lock' EXIT
trap 'exit 143' TERM INT HUP
echo "owner=Synaxis光照/followup pid=$$ since=$(date '+%F %T') task=RUN-065: ShardGates online check red/green, gate cases, two-process fixture, check and two strict modes" > $LOCK/owner
echo "$(date '+%F %T') taken: $(cat $LOCK/owner)" >> $TL
idle=0
while [[ ! -e $T/release ]]; do
  files=($T/queue/*.zsh(N)); next=${files[1]}
  if [[ -z $next ]]; then
    (( idle >= 600 )) && { echo "$(date '+%F %T') idle 10 min, leaving" >> $TL; break; }
    sleep 10; (( idle += 10 )); continue
  fi
  idle=0
  echo "$(date '+%F %T') start ${next:t}" >> $TL
  zsh $next > $T/out/${next:t:r}.stdout 2>&1
  echo "$(date '+%F %T') end ${next:t} rc=$?" >> $TL
  mv $next $T/done/
done
echo "$(date '+%F %T') holder done" >> $TL

# Gradle without / with the database variables; the console goes to $T/out/<tag>.log, the gist to stdout
gw() { local tag=$1; shift; env JAVA_TOOL_OPTIONS="$PROXY" ./gradlew "$@" $G > $T/out/$tag.log 2>&1; local rc=$?; gist $tag $rc; return $rc; }
gwdb() { local tag=$1; shift; env $DBV JAVA_TOOL_OPTIONS="$PROXY" ./gradlew "$@" $G > $T/out/$tag.log 2>&1; local rc=$?; gist $tag $rc; return $rc; }
gist() {
  echo "  gradle [$1] rc=$2"
  /usr/bin/grep -E 'tests completed|FAILED$|BUILD (SUCCESSFUL|FAILED)|^  *(org\.|java\.)[A-Za-z.]*(Error|Exception)|at [A-Za-z]+\.java:[0-9]+|error: |^e: ' $T/out/$1.log | /usr/bin/grep -v 'Picked up JAVA_TOOL_OPTIONS' | head -40 | sed 's/^/    /'
}

# JUnit XML of one class: counts, cases, failure messages, the test file lines in the traces
xmlsum() {
  local f=$RES/TEST-$1.xml
  [[ -f $f ]] || { echo "  (no result file for $1)"; return; }
  echo "  TEST-$1.xml: $(/usr/bin/grep -o -m1 'tests="[0-9]*" skipped="[0-9]*" failures="[0-9]*" errors="[0-9]*"' $f) $(/usr/bin/grep -o -m1 'timestamp="[^"]*"' $f)"
  /usr/bin/grep -o '<failure message="[^"]*"' $f | sed -e 's/<failure message=/    FAIL /' -e 's/&lt;/</g' -e 's/&gt;/>/g' -e 's/&quot;/"/g' -e "s/&#39;/'/g" -e 's/&amp;/\&/g' | cut -c1-700
  /usr/bin/grep -oE '\([A-Za-z]+Test\.java:[0-9]+\)' $f | sort | uniq -c | sed 's/^/    trace /'
  /usr/bin/grep -c AcquirableOwnershipException $f | sed 's/^/    AcquirableOwnershipException mentions: /'
}

# literal replacement in a file, fails loudly when the text is not there
subst() {
  OLD="$2" NEW="$3" perl -0pi -e 'my $o = $ENV{OLD}; my $n = $ENV{NEW}; my $c = s/\Q$o\E/$n/; die "no match in $ARGV for: $o\n" unless $c;' "$1"
}
restore() { git checkout HEAD -- "$@"; echo "  restored $*; git status --short -- server-app: [$(git status --short -- server-app | tr '\n' ' ')]"; }
showdiff() { git diff -U0 -- "$@" | /usr/bin/grep -E '^[-+][^-+]' | sed 's/^/    diff /'; }

unpaired = [i for i, c in pairs if c is None]

fb = sorted(c['first_batch_ms'] for c in completed)
