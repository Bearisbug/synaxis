# usage: g.sh <label> -- <gradle args...>   runs one Gradle in the integrate worktree, one at a time
label=$1; shift
[ "$1" = "--" ] && shift
source /tmp/m3e-post6a/env.sh
cd /Users/bug/Documents/Projects/Minecraft/synaxis-wt/m3e-integrate
while pgrep -f 'GradleWrapperMain|GradleDaemon|GradleMain' >/dev/null; do echo "waiting for another Gradle: $(pgrep -fl 'GradleWrapperMain|GradleDaemon|GradleMain' | cut -c1-80 | tr '\n' ';')"; sleep 20; done
log=/tmp/m3e-post6a/logs/$label.log
start=$(date +%s)
echo "# $label $(date '+%F %T') HEAD $(git rev-parse --short=10 HEAD) load1 $(sysctl -n vm.loadavg | awk '{print $2}') swap_free $(sysctl -n vm.swapusage | sed -E 's/.*free = ([0-9.]+M).*/\1/')" > $log
echo "# ./gradlew $* --no-daemon --init-script gradle/init/aliyun-mirror.init.gradle.kts --max-workers=4" >> $log
./gradlew "$@" --no-daemon --init-script gradle/init/aliyun-mirror.init.gradle.kts --max-workers=4 >> $log 2>&1
rc=$?
echo "# rc=$rc wall=$(( $(date +%s) - start ))s end $(date '+%T')" >> $log
tail -1 $log; grep -E 'BUILD (SUCCESSFUL|FAILED)|tests completed' $log | head -3
exit $rc

=== mut.pl
#!/usr/bin/perl
# usage: mut.pl <file> <old> <new>   literal, the old text must occur exactly once
use strict; use warnings;
my ($file, $old, $new) = @ARGV;
my $root = "/Users/bug/Documents/Projects/Minecraft/synaxis-wt/m3e-integrate/";
$file = $root . $file unless $file =~ m{^/};
open(my $in, '<', $file) or die "open $file: $!"; local $/; my $text = <$in>; close $in;
my $count = () = $text =~ /\Q$old\E/g;
die "mut: '$old' occurs $count times in $file\n" unless $count == 1;
$text =~ s/\Q$old\E/$new/;
open(my $out, '>', $file) or die; print $out $text; close $out;
my $line = 1 + (() = substr($text, 0, index($text, $new)) =~ /\n/g);
print "mut ok: $file:$line\n";
