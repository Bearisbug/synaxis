==================== variant.py
"""Applies one counter proof variant to the fixed listeners: one judgment switched off behind proof.never."""
import sys
from pathlib import Path

ROOT = Path("/Users/bug/Documents/Projects/Minecraft/synaxis-wt/m3fix-authority/src/main/java/io/github/bearisbug/synaxis/listener")
VARIANTS = {
    "border": ("PlayerPositionListener.java",
               "|| (border.inBounds(currentPosition) && !border.inBounds(packetPosition))",
               "|| (Boolean.getBoolean(\"proof.never\") && border.inBounds(currentPosition) && !border.inBounds(packetPosition))"),
    "cap": ("PlayerPositionListener.java",
            "|| currentPosition.distanceSquared(packetPosition) > 64 * 64",
            "|| (Boolean.getBoolean(\"proof.never\") && currentPosition.distanceSquared(packetPosition) > 64 * 64)"),
    "dig": ("PlayerActionListener.java",
            "if (outOfReach(player, blockPosition)) return true;",
            "if (Boolean.getBoolean(\"proof.never\") && outOfReach(player, blockPosition)) return true;"),
    "place": ("BlockPlacementListener.java",
              "|| PlayerActionListener.outOfReach(player, blockPosition)) {",
              "|| (Boolean.getBoolean(\"proof.never\") && PlayerActionListener.outOfReach(player, blockPosition))) {"),
}

file, old, new = VARIANTS[sys.argv[1]]
path = ROOT / file
text = path.read_text()
assert text.count(old) == 1, (file, old)
path.write_text(text.replace(old, new))
print(f"variant {sys.argv[1]}: {file}")

==================== summarize.py
"""Prints, for JUnit XML reports, the counts and every failure with its message and the test line it came from."""
import re
import sys
import xml.etree.ElementTree as ET

for path in sys.argv[1:]:
    root = ET.parse(path).getroot()
    name = root.get("name")
    print(f"{name}: tests={root.get('tests')} failures={root.get('failures')} errors={root.get('errors')} "
          f"skipped={root.get('skipped')} time={root.get('time')}")
    simple = name.rsplit(".", 1)[-1]
    for case in root.findall("testcase"):
        for kind in ("failure", "error"):
            node = case.find(kind)
            if node is None:
                continue
            message = (node.get("message") or "").splitlines()[0] if node.get("message") else ""
            text = node.text or ""
            line = re.search(re.escape(simple) + r"\.java:(\d+)", text)
            print(f"  {kind.upper()} {case.get('name')} at {simple}.java:{line.group(1) if line else '?'}: {message}")

==================== aggregate.py
"""Per module counts of a directory of JUnit XML reports (<dir>/<module>/TEST-*.xml), and every failing case."""
import sys
import xml.etree.ElementTree as ET
from pathlib import Path

base = Path(sys.argv[1])
total = [0, 0, 0, 0, 0]
for module in sorted(p for p in base.iterdir() if p.is_dir()):
    counts = [0, 0, 0, 0, 0]
    failed = []
    for report in sorted(module.glob("TEST-*.xml")):
        root = ET.parse(report).getroot()
        counts[0] += 1
        for i, key in enumerate(("tests", "failures", "errors", "skipped"), start=1):
            counts[i] += int(root.get(key, 0))
        for case in root.findall("testcase"):
            if case.find("failure") is not None or case.find("error") is not None:
                failed.append(f"{root.get('name')}.{case.get('name')}")
    total = [a + b for a, b in zip(total, counts)]
    print(f"  {base.name} {module.name}: classes={counts[0]} tests={counts[1]} failures={counts[2]} "
          f"errors={counts[3]} skipped={counts[4]}")
    for name in failed:
        print(f"    FAILED {name}")
print(f"  {base.name} all: classes={total[0]} tests={total[1]} failures={total[2]} errors={total[3]} skipped={total[4]}")
