#!/bin/zsh
# usage: gate.sh <label> <min free %> [own pids...]
# one reading: memory_pressure free %, swap (recorded only), load1, data-disk MB, shared containers, java/Gradle not in the own pid list, top 5 RSS
label=$1; minfree=$2; shift 2; own=" $* "
now=$(date '+%Y-%m-%d %H:%M:%S')
freepct=$(memory_pressure | sed -nE 's/System-wide memory free percentage: ([0-9]+)%/\1/p')
swap=$(sysctl -n vm.swapusage | sed -E 's/total = ([0-9.]+)M +used = ([0-9.]+)M +free = ([0-9.]+)M.*/total=\1MB used=\2MB free=\3MB/')
load=$(sysctl -n vm.loadavg | awk '{print $2}')
disk=$(df -m /System/Volumes/Data | awk 'NR==2{print $4}')
ctr=$(docker ps --format '{{.Names}}' | grep -cE '^synaxis-(pg|valkey)$')
foreign=""
for p in $(pgrep -f 'java|gradle'); do
  cmd=$(ps -o comm= -p $p 2>/dev/null) || continue
  case "$cmd" in *java*|*gradle*) ;; *) continue;; esac
  case "$own" in *" $p "*) continue;; esac
  cwd=$(lsof -a -d cwd -p $p -Fn 2>/dev/null | grep ^n | cut -c2-)
  foreign="$foreign $p($cwd)"
done
top=$(ps -axo rss=,comm= | sort -n -r | head -5 | awk '{n=$2; sub(".*/","",n); printf "%s=%dMB ", n, $1/1024}')
ok=yes
(( freepct >= minfree )) || ok=no
[ -z "$foreign" ] || ok=no
(( disk >= 5120 )) || ok=no
echo "$now [$label] mem_free=${freepct}% (>= ${minfree}%) swap: $swap load1=$load disk_free=${disk}MB containers=$ctr foreign_java_gradle=[${foreign# }] top5: $top met=$ok"
[ $ok = yes ]

=== <L>/monitor2.sh
#!/bin/zsh
# usage: monitor2.sh <out> <every s>: time, memory free %, swap free, load1, top 5 by RSS and by CPU, until killed
out=$1; every=$2
while true; do
  now=$(date '+%H:%M:%S')
  freepct=$(memory_pressure | sed -nE 's/System-wide memory free percentage: ([0-9]+)%/\1/p')
  swapfree=$(sysctl -n vm.swapusage | sed -E 's/.*free = ([0-9.]+)M.*/\1/')
  load=$(sysctl -n vm.loadavg | awk '{print $2}')
  top=$(ps -axo rss=,comm= | sort -n -r | head -5 | awk '{n=$2; sub(".*/","",n); printf "%s=%dMB ", n, $1/1024}')
  cpu=$(ps -Ao pcpu=,comm= -r | head -5 | awk '{n=$2; sub(".*/","",n); printf "%s=%s%% ", n, $1}')
  echo "$now mem_free=${freepct}% swap_free=${swapfree}MB load1=$load rss_top5: $top cpu_top5: $cpu" >> $out
  sleep $every
done

import json, sys, datetime
# usage: slowticks.py <json of jfr print --json --events synaxis.ServerTickTime,jdk.GCPhasePause>
d = json.load(open(sys.argv[1]))
ev = d['recording']['events']
def ts(s):
    return datetime.datetime.fromisoformat(s.replace('Z', '+00:00')[:26] + '+00:00' if '.' in s else s.replace('Z', '+00:00'))
def dur(s):  # ISO-8601 duration like PT0.088813S
    s = s[2:]
    total = 0.0
    num = ''
    for ch in s:
        if ch.isdigit() or ch == '.': num += ch
        else:
            v = float(num); num = ''
            total += v * {'H': 3600, 'M': 60, 'S': 1}[ch]
    return total
ticks, gcs = [], []
for e in ev:
    t = e['type']; v = e['values']
    start = ts(v['startTime']); d_ = dur(v['duration'])
    if t == 'synaxis.ServerTickTime': ticks.append((start, d_, v))
    elif t == 'jdk.GCPhasePause': gcs.append((start, d_, v))
slow = [x for x in ticks if x[1] > 0.050]
print('ServerTickTime events', len(ticks), 'over 50 ms', len(slow), 'GCPhasePause', len(gcs), 'over 50 ms', sum(1 for g in gcs if g[1] > 0.050))
for s, d_, v in slow:
    end = s + datetime.timedelta(seconds=d_)
    over = [(g[0], g[1], g[2].get('name')) for g in gcs if g[0] < end and g[0] + datetime.timedelta(seconds=g[1]) > s]
    fields = {k: v[k] for k in v if k not in ('startTime', 'duration', 'eventThread', 'stackTrace')}
    print(f'slow tick {s.isoformat()} {d_*1000:.3f} ms {fields} overlapping GC pauses:', [(o[0].isoformat(), round(o[1]*1000, 3), o[2]) for o in over])

import glob, json, math, sys
# usage: transfers084.py <t084 dir>: the two ledgers' transfer events, paired by transfer_ref, and the bot-side transfer figures
D = sys.argv[1]
def pct(values, p):
    # nearest rank
    if not values: return None
    v = sorted(values)
    return v[max(0, math.ceil(p / 100 * len(v)) - 1)]
events = {}
for shard in ('r0-c1', 'r1-c1'):
    for f in glob.glob(f'{D}/work-{shard}/logs/events-*.jsonl'):
        for line in open(f):
            e = json.loads(line)
            e['ledger'] = shard
            events.setdefault(e['event'], []).append(e)
print('ledger events by type:', {k: len(v) for k, v in sorted(events.items())})
for shard in ('r0-c1', 'r1-c1'):
    print(f'  {shard}:', {k: sum(1 for e in v if e['ledger'] == shard) for k, v in sorted(events.items())})
issued, completed = events.get('transfer_issued', []), events.get('transfer_completed', [])
print('transfer_issued', len(issued), 'transfer_completed', len(completed),
      'completed / issued =', round(len(completed) / len(issued), 5) if issued else None)
by_ref = {}
for c in completed: by_ref.setdefault(c['transfer_ref'], []).append(c)
unpaired = [i for i in issued if i['transfer_ref'] not in by_ref]
dup = [r for r, cs in by_ref.items() if len(cs) > 1]
iss_refs = {i['transfer_ref'] for i in issued}
orphan = [c for c in completed if c['transfer_ref'] not in iss_refs]
print('paired by transfer_ref:', len(issued) - len(unpaired), 'issued without completed:', len(unpaired),
      'completed without issued:', len(orphan), 'refs completed twice:', len(dup))
print('issued in the source ledger:', all(i['ledger'] == i['source_shard'] for i in issued),
      'completed in the target ledger:', all(c['ledger'] == c['target_shard'] for c in completed))
dirs = {}
for i in issued: dirs[(i['source_shard'], i['target_shard'])] = dirs.get((i['source_shard'], i['target_shard']), 0) + 1
print('issued by direction:', {f'{a}->{b}': n for (a, b), n in sorted(dirs.items())})
fb = [c['first_batch_ms'] for c in completed if c.get('first_batch_ms') is not None]
if fb:
    print('first_batch_ms: count', len(fb), 'min', min(fb), 'p50', pct(fb, 50), 'p95', pct(fb, 95), 'p99', pct(fb, 99), 'max', max(fb),
          'P95 < 5000:', pct(fb, 95) < 5000)
