## the management client (standard library only)
    """Management interface client, standard library only: WebSocket upgrade with the bearer token,
    one masked text frame, skips notifications without an id, prints the reply that has the id.
    usage: mgmt.py <port> <method> [<params as JSON>]; the token is read from the file in MGMT_TOKEN_FILE."""
    import base64
    import json
    import os
    import socket
    import struct
    import sys
    
    port, method = int(sys.argv[1]), sys.argv[2]
    params = sys.argv[3] if len(sys.argv) > 3 else None
    with open(os.environ["MGMT_TOKEN_FILE"]) as f:
        token = f.read().strip()
    
    s = socket.create_connection(("127.0.0.1", port), timeout=15)
    key = base64.b64encode(os.urandom(16)).decode()
    s.sendall((f"GET / HTTP/1.1\r\nHost: 127.0.0.1:{port}\r\nUpgrade: websocket\r\nConnection: Upgrade\r\n"
               f"Sec-WebSocket-Key: {key}\r\nSec-WebSocket-Version: 13\r\nAuthorization: Bearer {token}\r\n\r\n").encode())
    buf = b""
    while b"\r\n\r\n" not in buf:
        chunk = s.recv(4096)
        if not chunk:
            sys.exit(f"closed during the handshake: {buf[:200]!r}")
        buf += chunk
    head, buf = buf.split(b"\r\n\r\n", 1)
    status = head.split(b"\r\n")[0].decode()
    if " 101 " not in status:
        sys.exit(f"handshake answered {status}")
    
    
    def send(opcode, payload):
        mask = os.urandom(4)
        n = len(payload)
        if n < 126:
            header = bytes([0x80 | opcode, 0x80 | n])
        elif n < 65536:
            header = bytes([0x80 | opcode, 0x80 | 126]) + struct.pack(">H", n)
        else:
            header = bytes([0x80 | opcode, 0x80 | 127]) + struct.pack(">Q", n)
        s.sendall(header + mask + bytes(b ^ mask[i % 4] for i, b in enumerate(payload)))
    
    
    def take(n):
        global buf
        while len(buf) < n:
            chunk = s.recv(65536)
            if not chunk:
                sys.exit("connection closed before the reply")
            buf += chunk
        out, buf = buf[:n], buf[n:]
        return out
    
    
    message = {"jsonrpc": "2.0", "id": 1, "method": method}
    if params is not None:
        message["params"] = json.loads(params)
    send(0x1, json.dumps(message, separators=(",", ":")).encode())
    
    text = b""
    while True:
        b0, b1 = take(2)
        opcode, fin, n = b0 & 0x0F, b0 & 0x80, b1 & 0x7F
        if n == 126:
            n = struct.unpack(">H", take(2))[0]
        elif n == 127:
            n = struct.unpack(">Q", take(8))[0]
        payload = take(n)
        if opcode == 0x8:
            sys.exit(f"closed by the server: {payload!r}")
        if opcode == 0x9:
            send(0xA, payload)
            continue
        if opcode in (0x1, 0x0):
            text += payload
            if not fin:
                continue
            reply = json.loads(text)
            text = b""
            if "id" in reply:
                print(json.dumps(reply, separators=(",", ":"), ensure_ascii=False))
                break
    send(0x8, struct.pack(">H", 1000))
    s.close()
