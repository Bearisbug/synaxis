#!/bin/zsh
# usage: gate.sh <label> <min free %> [own pids...]
# One reading of the start gate (docs/TEST.md TC-WORLD-081 precondition with the 35% of TC-WORLD-082/083):
# memory_pressure free % >= min; every java process from pgrep -fl java other than the round's own averages <= 5% CPU over 30 s
# (ps -o %cpu= -p <pid>, 3 samples 15 s apart); /tmp/synaxis-heavy.lock held by this round; df -m /System/Volumes/Data >= 5,120 MB;
# both shared containers up. vm.swapusage, load1 and the top 5 processes are recorded only.
source ${0:A:h}/env.sh
label=$1; minfree=$2; shift 2; own=" $* "
now=$(date '+%Y-%m-%d %H:%M:%S')
freepct=$(memory_pressure | sed -nE 's/System-wide memory free percentage: ([0-9]+)%/\1/p')
swap=$(sysctl -n vm.swapusage | sed -E 's/total = ([0-9.]+)M +used = ([0-9.]+)M +free = ([0-9.]+)M.*/total=\1MB used=\2MB free=\3MB/')
load=$(sysctl -n vm.loadavg | awk '{print $2}')
disk=$(df -m /System/Volumes/Data | awk 'NR==2{print $4}')
ctr=$(docker ps --format '{{.Names}}' | /usr/bin/grep -cE '^synaxis-(pg|valkey)$')
lockok=no
/usr/bin/grep -q '^owner=Synaxis光照/run063 pid=39601 ' /tmp/synaxis-heavy.lock/owner 2>/dev/null && lockok=yes
toprss=$(ps -axo rss=,comm= | sort -n -r | head -5 | awk '{n=$2; sub(".*/","",n); printf "%s=%dMB ", n, $1/1024}')
topcpu=$(ps -Ao pcpu=,comm= -r | head -5 | awk '{n=$2; sub(".*/","",n); printf "%s=%s%% ", n, $1}')
# java processes listed by pgrep -fl java (the executable is java), the round's own left out
cands=()
for p in $(pgrep -f java); do
  case "$own" in *" $p "*) continue;; esac
  comm=$(ps -o comm= -p $p 2>/dev/null) || continue
  case "${comm:t}" in java) cands+=$p;; esac
done
typeset -A c1 c2 c3
for p in $cands; do c1[$p]=$(ps -o %cpu= -p $p 2>/dev/null | tr -d ' '); done
(( ${#cands} > 0 )) && sleep 15
for p in $cands; do c2[$p]=$(ps -o %cpu= -p $p 2>/dev/null | tr -d ' '); done
(( ${#cands} > 0 )) && sleep 15
for p in $cands; do c3[$p]=$(ps -o %cpu= -p $p 2>/dev/null | tr -d ' '); done
javaok=yes; jdetail=""
for p in $cands; do
  a=${c1[$p]:-0}; b=${c2[$p]:-0}; c=${c3[$p]:-0}
  mean=$(echo "$a $b $c" | awk '{printf "%.2f", ($1+$2+$3)/3}')
  ok=$(echo "$mean" | awk '{print ($1 <= 5.0) ? "yes" : "no"}')
  [ $ok = yes ] || javaok=no
  cwd=$(lsof -a -d cwd -p $p -Fn 2>/dev/null | /usr/bin/grep ^n | cut -c2-)
  args=$(ps -o args= -p $p 2>/dev/null | sed -E 's/ -(cp|classpath) [^ ]+//' | cut -c1-160)
  jdetail="$jdetail
    java pid=$p cwd=$cwd cpu%=[$a $b $c] mean=$mean (<= 5) ok=$ok args: $args"
done
met=yes
(( freepct >= minfree )) || met=no
[ $javaok = yes ] || met=no
[ $lockok = yes ] || met=no
(( disk >= 5120 )) || met=no
(( ctr == 2 )) || met=no
echo "$now [$label] mem_free=${freepct}% (>= ${minfree}%) swap: $swap load1=$load disk_free=${disk}MB (>= 5120) containers=$ctr lock_held=$lockok other_java=${#cands} java_cpu_ok=$javaok top5_rss: $toprss top5_cpu: $topcpu met=$met$jdetail"
[ $met = yes ]

=== <L>/monitor.sh
#!/bin/zsh
# usage: monitor.sh <out> <every s>: time, memory free %, swap free, load1, top 5 by RSS and by CPU, until killed
out=$1; every=$2
while true; do
  now=$(date '+%H:%M:%S')
  freepct=$(memory_pressure | sed -nE 's/System-wide memory free percentage: ([0-9]+)%/\1/p')
  swapfree=$(sysctl -n vm.swapusage | sed -E 's/.*free = ([0-9.]+)M.*/\1/')
  load=$(sysctl -n vm.loadavg | awk '{print $2}')
  top=$(ps -axo rss=,comm= | sort -n -r | head -5 | awk '{n=$2; sub(".*/","",n); printf "%s=%dMB ", n, $1/1024}')
  cpu=$(ps -Ao pcpu=,comm= -r | head -5 | awk '{n=$2; sub(".*/","",n); printf "%s=%s%% ", n, $1}')
  echo "$now mem_free=${freepct}% swap_free=${swapfree}MB load1=$load rss_top5: $top cpu_top5: $cpu" >> $out
  sleep $every
done

#!/bin/zsh
# RUN-063 job: heavy-work lock (taken once, released by the EXIT trap at the top level of this script after the cleanup),
# build, bake, the two databases, one shared 90 min gate window for TC-WORLD-082 then TC-WORLD-083, the analysis, then (after
# the go-cleanup flag or 60 min) the cleanup. Output: stdout, redirected by the launcher.
source ${0:A:h}/env.sh
LOCK=/tmp/synaxis-heavy.lock
G=$L/gates.txt
say() { echo "$(date '+%F %T') $*"; }

say "job start, pid $$; waiting for the lock $LOCK"
while ! mkdir $LOCK 2>/dev/null; do
  owner=$(cat $LOCK/owner 2>/dev/null)
  opid=$(echo "$owner" | sed -nE 's/.* pid=([0-9]+).*/\1/p')
  say "lock held by another session: owner file [$owner]"
  if [ -n "$opid" ] && ! kill -0 $opid 2>/dev/null; then
    say "the owner pid $opid is gone: stale lock, owner recorded above, removing it"
    rm -rf $LOCK
    continue
  fi
  sleep 60
done
trap 'if /usr/bin/grep -q "^owner=Synaxis光照/run063 pid=39601 " /tmp/synaxis-heavy.lock/owner 2>/dev/null; then rm -rf /tmp/synaxis-heavy.lock; echo "$(date "+%F %T") lock released by the EXIT trap; /tmp/synaxis-heavy.lock exists afterwards: $([ -e /tmp/synaxis-heavy.lock ] && echo yes || echo no)"; else echo "$(date "+%F %T") EXIT trap: the lock is not this round'"'"'s, left alone"; fi' EXIT
trap 'exit 143' TERM
trap 'exit 130' INT
since=$(date '+%Y-%m-%dT%H:%M:%S%z')
echo "owner=Synaxis光照/run063 pid=39601 since=$since task=RUN-063: TC-WORLD-082/083 500-bot tiers" > $LOCK/owner
say "lock taken (mkdir succeeded)"
sleep 1
if [ -d $LOCK ] && [ "$(cat $LOCK/owner)" = "owner=Synaxis光照/run063 pid=39601 since=$since task=RUN-063: TC-WORLD-082/083 500-bot tiers" ] && [ $(wc -l < $LOCK/owner) -eq 1 ]; then
  say "lock self-check ok: the directory is there and owner is this round's one line: $(cat $LOCK/owner)"
else
  say "lock self-check FAILED: $(ls -ld $LOCK 2>&1) owner [$(cat $LOCK/owner 2>&1)]"; exit 1
fi

# baselines for the cleanup
ls -1 $TMPDIR > $L/tmpdir-before.txt
ls -1 /tmp > $L/slash-tmp-before.txt
ls -1 "$HOME/Library/Application Support/kotlin/daemon" > $L/kotlin-daemon-before.txt 2>/dev/null
say "baseline: df -m /System/Volumes/Data available $(df -m /System/Volumes/Data | awk 'NR==2{print $4}') MB; databases: $(docker exec synaxis-pg psql -U postgres -tAc 'select datname from pg_database order by 1' | tr '\n' ' ')"

# build
say "build: installDist of server-app, loadtest, baker"
$L/gradle.sh install :server-app:installDist :loadtest:installDist :baker:installDist || { say "build failed, see $L/gradle-install.log"; tail -30 $L/gradle-install.log; exit 1; }
say "build done: $(/usr/bin/grep -E 'BUILD|actionable|exit' $L/gradle-install.log | tr '\n' ' ')"

# bake the 4k test map
say "bake start"
cd $L
/usr/bin/time -l $R/baker/build/install/baker/bin/baker bake --map-id synth --source generator:synthetic:7 --out $L/maps --logs $L/bakelogs --threads 4 > $L/bake.log 2>&1
rc=$?
say "bake exit $rc"
[ $rc -eq 0 ] || { tail -30 $L/bake.log; exit 1; }
[ -d $L/maps/synth/$B ] || { say "no release $L/maps/synth/$B"; ls -la $L/maps/synth; exit 1; }

# the password of synaxis-pg (never printed) and the two tiers' databases
umask 077
docker inspect synaxis-pg --format '{{range .Config.Env}}{{println .}}{{end}}' | sed -n 's/^POSTGRES_PASSWORD=//p' > $L/pgpass
umask 022
[ -s $L/pgpass ] || { say "no password read from docker inspect"; exit 1; }
for t in t082 t083; do
  $L/setup_tier.sh $t $L/shards-$t.json || { say "setup $t failed"; cat $L/$t/migrate.log 2>/dev/null | tail -20; exit 1; }
  say "setup $t: $(cat $L/$t/migrate.log | /usr/bin/grep -E '^migrate exit|\|t' | tr '\n' ' ')"
done

# one shared gate window: a reading, then every 5 min until met or until the next reading would be past 90 min
wstart=$(date +%s); deadline=$((wstart + 90 * 60))
echo "# window $(date -r $wstart '+%F %T') to $(date -r $deadline '+%F %T'); gate: memory free >= 35%, every other java process <= 5% CPU averaged over 30 s, lock held by this round, data disk >= 5,120 MB, both shared containers" >> $G
wait_gate() { # label
  local t0
  while true; do
    t0=$(date +%s)
    $L/gate.sh "$1" 35 >> $G && return 0
    (( t0 + 300 > deadline )) && { echo "# $1: gate not met by the end of the window $(date '+%F %T'), blocked" >> $G; return 1; }
    sleep $(( t0 + 300 - $(date +%s) ))
  done
}
ran082=no; ran083=no
if wait_gate "082 start"; then
  say "082 start"
  $L/run_m3e.sh t082 r0-c1 25621 19621 500 walk 600
  echo "# t082 runner exit $? $(date '+%F %T')" >> $G
  ran082=yes
  say "082 done; left processes of 082: $(for p in $(cat $L/t082/pids.txt); do kill -0 $p 2>/dev/null && echo -n "$p "; done)"
  if wait_gate "083 start (after 082)"; then
    say "083 start"
    $L/run_m3e.sh t083 town-c1 25621 19621 500 town 600
    echo "# t083 runner exit $? $(date '+%F %T')" >> $G
    ran083=yes
    say "083 done; left processes of 083: $(for p in $(cat $L/t083/pids.txt); do kill -0 $p 2>/dev/null && echo -n "$p "; done)"
  fi
fi
echo "# window done $(date '+%F %T') ran082=$ran082 ran083=$ran083" >> $G
say "gate window done: ran082=$ran082 ran083=$ran083"

# analysis
mkdir -p $L/evidence
[ $ran082 = yes ] && $L/analyze_m3e.sh t082 > $L/evidence/082.txt 2>&1
[ $ran083 = yes ] && $L/analyze_m3e.sh t083 > $L/evidence/083.txt 2>&1
say "analysis written; waiting for $L/go-cleanup (at most 60 min)"
touch $L/analysis.done
w=0
while [ ! -e $L/go-cleanup ] && (( w < 3600 )); do sleep 10; w=$((w + 10)); done
say "cleanup start (go flag: $([ -e $L/go-cleanup ] && echo yes || echo 'no, 60 min passed'))"

# cleanup: processes, ports, databases, new temp entries, <L>; the lock goes last (EXIT trap)
for t in t082 t083; do
  [ -f $L/$t/pids.txt ] || continue
  for p in $(cat $L/$t/pids.txt); do
    if kill -0 $p 2>/dev/null; then say "$t pid $p still alive: $(ps -o comm= -p $p), SIGTERM"; kill -TERM $p; sleep 2; kill -0 $p 2>/dev/null && kill -KILL $p; fi
  done
  say "$t pids $(cat $L/$t/pids.txt | tr '\n' ' '): alive now: $(for p in $(cat $L/$t/pids.txt); do kill -0 $p 2>/dev/null && echo -n "$p "; done)"
done
say "processes whose command line names <L>: $(pgrep -f "$L" | tr '\n' ' ')"
say "listeners on 25621 and 19621: $(lsof -nP -iTCP:25621 -iTCP:19621 -sTCP:LISTEN 2>/dev/null | tail -n +2 | wc -l | tr -d ' ')"
for t in t082 t083; do docker exec synaxis-pg psql -U postgres -qc "DROP DATABASE IF EXISTS run063_$t"; done
say "databases after the drop: $(docker exec synaxis-pg psql -U postgres -tAc 'select datname from pg_database order by 1' | tr '\n' ' ')"
ls -1 $TMPDIR > $L/tmpdir-after.txt
ls -1 /tmp > $L/slash-tmp-after.txt
ls -1 "$HOME/Library/Application Support/kotlin/daemon" > $L/kotlin-daemon-after.txt 2>/dev/null
newtmp=$(comm -13 <(sort $L/tmpdir-before.txt) <(sort $L/tmpdir-after.txt))
say "new entries in \$TMPDIR since the round started: $(echo $newtmp | tr '\n' ' ')"
for e in ${(f)newtmp}; do say "  $e: $(stat -f '%SB %N' -t '%F %T' $TMPDIR/$e)"; done
say "new entries in /tmp: $(comm -13 <(sort $L/slash-tmp-before.txt) <(sort $L/slash-tmp-after.txt) | tr '\n' ' ')"
say "new entries in kotlin/daemon: $(comm -13 <(sort $L/kotlin-daemon-before.txt) <(sort $L/kotlin-daemon-after.txt) | tr '\n' ' ')"
for e in ${(f)newtmp}; do
  case "$e" in
    kotlin-daemon.*.log|synaxis-loadtest-*.jfr|junit-*)
      if /usr/bin/grep -qs "synaxis-wt/run-063\|run063" $TMPDIR/$e 2>/dev/null || [[ $e == synaxis-loadtest-*.jfr ]]; then say "  deleting $e (this round's)"; rm -rf $TMPDIR/$e; else say "  leaving $e (does not name this round)"; fi;;
  esac
done
before=$(df -m /System/Volumes/Data | awk 'NR==2{print $4}')
say "df -m available before deleting <L>: $before MB; <L> size $(du -sm $L | cut -f1) MB"
cd /
rm -rf $L
after=$(df -m /System/Volumes/Data | awk 'NR==2{print $4}')
say "df -m available after deleting <L>: $after MB (difference $((after - before)) MB); <L> exists: $([ -e $L ] && echo yes || echo no)"
say "job end"
exit 0
