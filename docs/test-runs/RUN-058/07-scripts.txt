#!/bin/zsh
# usage: gate.sh <label> <min free %> [own pids...]
# one reading: memory_pressure free %, swap (recorded only), load1, data-disk MB, java/Gradle not in the own pid list, top 5 RSS and CPU
label=$1; minfree=$2; shift 2; own=" $* "
now=$(date '+%Y-%m-%d %H:%M:%S')
freepct=$(memory_pressure | sed -nE 's/System-wide memory free percentage: ([0-9]+)%/\1/p')
swap=$(sysctl -n vm.swapusage | sed -E 's/total = ([0-9.]+)M +used = ([0-9.]+)M +free = ([0-9.]+)M.*/total=\1MB used=\2MB free=\3MB/')
load=$(sysctl -n vm.loadavg | awk '{print $2}')
disk=$(df -m /System/Volumes/Data | awk 'NR==2{print $4}')
foreign=""
for p in $(pgrep -f 'java|gradle'); do
  cmd=$(ps -o comm= -p $p 2>/dev/null) || continue
  case "$cmd" in *java*|*gradle*) ;; *) continue;; esac
  case "$own" in *" $p "*) continue;; esac
  cwd=$(lsof -a -d cwd -p $p -Fn 2>/dev/null | grep ^n | cut -c2-)
  foreign="$foreign $p($cwd)"
done
top=$(ps -axo rss=,comm= | sort -n -r | head -5 | awk '{n=$2; sub(".*/","",n); printf "%s=%dMB ", n, $1/1024}')
cpu=$(ps -Ao pcpu=,comm= -r | head -5 | awk '{n=$2; sub(".*/","",n); printf "%s=%s%% ", n, $1}')
ok=yes
(( freepct >= minfree )) || ok=no
[ -z "$foreign" ] || ok=no
(( disk >= 5120 )) || ok=no
echo "$now [$label] mem_free=${freepct}% (>= ${minfree}%) swap: $swap load1=$load disk_free=${disk}MB foreign_java_gradle=[${foreign# }] rss_top5: $top cpu_top5: $cpu met=$ok"
[ $ok = yes ]

=== <L>/wait_gate.sh
#!/bin/zsh
# usage: wait_gate.sh <label> <min free %> <out>: a reading now, then every 5 min while not met, at most 90 min (19 readings)
label=$1; minfree=$2; out=$3; dir=${0:h}
for i in {0..18}; do
  $dir/gate.sh "$label" $minfree >> $out && exit 0
  (( i < 18 )) && sleep 300
done
echo "$(date '+%F %T') [$label] not met within 90 min: blocked" >> $out
exit 1

#!/bin/zsh
# usage: monitor.sh <out> <every s> <own pid file>: time, memory free %, swap free, load1, top 5 by RSS and CPU, other sessions' heavy
# processes (java/Gradle not in the own pid file, qemu = Android emulator, launchd_sim = iOS simulator, xcodebuild), until killed
out=$1; every=$2; ownf=$3
while true; do
  now=$(date '+%H:%M:%S')
  freepct=$(memory_pressure | sed -nE 's/System-wide memory free percentage: ([0-9]+)%/\1/p')
  swapfree=$(sysctl -n vm.swapusage | sed -E 's/.*free = ([0-9.]+)M.*/\1/')
  load=$(sysctl -n vm.loadavg | awk '{print $2}')
  top=$(ps -axo rss=,comm= | sort -n -r | head -5 | awk '{n=$2; sub(".*/","",n); printf "%s=%dMB ", n, $1/1024}')
  cpu=$(ps -Ao pcpu=,comm= -r | head -5 | awk '{n=$2; sub(".*/","",n); printf "%s=%s%% ", n, $1}')
  own=" $(tr '\n' ' ' < $ownf) "
  foreign=""
  for p in $(pgrep -f 'java|gradle'); do
    cmd=$(ps -o comm= -p $p 2>/dev/null) || continue
    case "$cmd" in *java*|*gradle*) ;; *) continue;; esac
    case "$own" in *" $p "*) continue;; esac
    args=$(ps -o args= -p $p 2>/dev/null | cut -c1-60)
    case "$args" in *jcmd*|*jdk.jfr*|*sun.tools*) continue;; esac
    foreign="$foreign $p"
  done
  heavy="qemu=$(pgrep -f qemu-system | wc -l | tr -d ' ') launchd_sim=$(pgrep -x launchd_sim | wc -l | tr -d ' ') xcodebuild=$(pgrep -x xcodebuild | wc -l | tr -d ' ') foreign_java_gradle=[${foreign# }]"
  echo "$now mem_free=${freepct}% swap_free=${swapfree}MB load1=$load $heavy rss_top5: $top cpu_top5: $cpu" >> $out
  sleep $every
done

=== <L>/gcticks.py
import datetime, json, re, sys
# usage: gcticks.py <json of jfr print --json --events synaxis.ServerTickTime,jdk.GCPhasePause,jdk.G1GarbageCollection> <gc.log>
# ticks over 50 ms (synaxis.ServerTickTime) against the G1 pauses they overlap, from JFR (jdk.GCPhasePause) and from the -Xlog:gc* file
TS = re.compile(r'(\d{4}-\d\d-\d\dT\d\d:\d\d:\d\d)(?:\.(\d+))?(Z|[+-]\d\d:?\d\d)?')
def ts(s):
    m = TS.match(s)
    frac = (m.group(2) or '0')[:6].ljust(6, '0')
    off = m.group(3) or 'Z'
    off = '+00:00' if off == 'Z' else (off if ':' in off else off[:3] + ':' + off[3:])
    return datetime.datetime.fromisoformat(f'{m.group(1)}.{frac}{off}')
def dur(s):  # ISO-8601 duration like PT0.088813S
    total, num = 0.0, ''
    for ch in s[2:]:
        if ch.isdigit() or ch == '.': num += ch
        else: total += float(num) * {'H': 3600, 'M': 60, 'S': 1}[ch]; num = ''
    return total
def local(t): return t.astimezone().strftime('%H:%M:%S.%f')[:-3]
def pct(v, p):
    v = sorted(v); return v[max(0, -(-p * len(v) // 100) - 1)] if v else None
ev = json.load(open(sys.argv[1]))['recording']['events']
ticks, jgc, g1type = [], [], {}
for e in ev:
    t, v = e['type'], e['values']
    if t == 'synaxis.ServerTickTime': ticks.append((ts(v['startTime']), dur(v['duration'])))
    elif t == 'jdk.GCPhasePause': jgc.append((ts(v['startTime']), dur(v['duration']), v.get('gcId')))
    elif t == 'jdk.G1GarbageCollection': g1type[v.get('gcId')] = v.get('type')
PAUSE = re.compile(r'^\[([^\]]+)\]\[[^\]]+\]\[info\s*\]\[gc\s*\] GC\((\d+)\) (Pause .*?) (\d+M->\d+M\(\d+M\)) ([\d.]+)ms$')
lg = []
for line in open(sys.argv[2]):
    m = PAUSE.match(line.rstrip())
    if m:
        end, p = ts(m.group(1)), float(m.group(5)) / 1000
        lg.append((end - datetime.timedelta(seconds=p), p, int(m.group(2)), m.group(3), m.group(4)))
slow = [x for x in ticks if x[1] > 0.050]
print(f'JFR synaxis.ServerTickTime: {len(ticks)} ticks, over 50 ms {len(slow)}, max {max(d for _, d in ticks)*1000:.3f} ms')
jd = [d * 1000 for _, d, _ in jgc]
print(f'JFR jdk.GCPhasePause: {len(jgc)} pauses, over 50 ms {sum(1 for d in jd if d > 50)}, P50 {pct(jd, 50):.3f} ms, max {max(jd):.3f} ms, sum {sum(jd):.1f} ms' if jd else 'JFR jdk.GCPhasePause: 0 pauses')
ld = [p * 1000 for _, p, _, _, _ in lg]
kinds = {}
for _, p, _, k, _ in lg: kinds.setdefault(k, []).append(p * 1000)
print(f'gc.log pauses: {len(lg)}, over 50 ms {sum(1 for d in ld if d > 50)}, P50 {pct(ld, 50):.3f} ms, P99 {pct(ld, 99):.3f} ms, max {max(ld):.3f} ms, sum {sum(ld):.1f} ms' if ld else 'gc.log pauses: 0')
for k, v in sorted(kinds.items()): print(f'  {k}: {len(v)}, max {max(v):.3f} ms')
def over(s, d, pauses):
    e = s + datetime.timedelta(seconds=d)
    return [p for p in pauses if p[0] < e and p[0] + datetime.timedelta(seconds=p[1]) > s]
def nearest(s, pauses):
    return min(pauses, key=lambda p: abs((p[0] - s).total_seconds()), default=None)
print('slow ticks (local time; tick start, duration) and the pauses they overlap:')
for s, d in slow:
    oj = over(s, d, jgc); ol = over(s, d, lg)
    js = [f'JFR gcId {p[2]} {g1type.get(p[2], "?")} {local(p[0])} {p[1]*1000:.3f} ms' for p in oj]
    ls = [f'gc.log GC({p[2]}) {p[3]} {p[4]} {local(p[0])} {p[1]*1000:.3f} ms' for p in ol]
    line = f'  tick {local(s)} {d*1000:.3f} ms: ' + ('; '.join(js + ls) if js or ls else 'no overlapping pause')
    if not ol:
        n = nearest(s, lg)
        if n: line += f' (nearest gc.log pause GC({n[2]}) {n[3]} at {local(n[0])}, {(n[0]-s).total_seconds():+.3f} s, {n[1]*1000:.3f} ms)'
    print(line)
print('gc.log pauses of 30 ms or more and the longest tick overlapping each:')
for p in sorted(lg, key=lambda p: p[0]):
    if p[1] >= 0.030:
        ot = over(p[0], p[1], ticks)
        m = max((d for _, d in ot), default=None)
        print(f'  GC({p[2]}) {p[3]} {p[4]} {local(p[0])} {p[1]*1000:.3f} ms: ' + (f'{len(ot)} tick(s), longest {m*1000:.3f} ms' if ot else 'no tick overlapping'))
