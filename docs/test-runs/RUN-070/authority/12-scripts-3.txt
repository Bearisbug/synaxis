"""Applies one counter proof variant to the working tree: a pause put in, a line taken out or one judgment switched off."""
import sys
from pathlib import Path

WT = Path("/Users/bug/Documents/Projects/Minecraft/synaxis-wt/m3fix-authority")
BOT = WT / "loadtest/src/main/java/io/github/bearisbug/synaxis/loadtest/Bot.java"
PPL = WT / "src/main/java/io/github/bearisbug/synaxis/listener/PlayerPositionListener.java"
VARIANTS = {
    # 200 ms between the configuration ack and the switch of state, where the moving thread can send
    "pause": (BOT,
              "send(ConnectionState.PLAY, new ClientConfigurationAckPacket());",
              "send(ConnectionState.PLAY, new ClientConfigurationAckPacket());"
              " LockSupport.parkNanos(TimeUnit.MILLISECONDS.toNanos(200));"),
    # the teleport no longer resets where the bot takes the server to have it
    "reset": (BOT,
              "                    serverPosition = new double[]{teleport.position().x(), teleport.position().y(), teleport.position().z()};\n",
              ""),
    "bot": (BOT,
            "            final double[] next = toward(serverPosition, at);",
            "            final double[] next = Boolean.getBoolean(\"proof.never\") ? toward(serverPosition, at) : at;"),
    "rounding": (PPL,
                 "        final double minX = Math.floor(border.centerX() - radius), maxX = Math.ceil(border.centerX() + radius);",
                 "        final double minX = border.centerX() - radius, maxX = border.centerX() + radius;"),
    "rounding-z": (PPL,
                   "        final double minZ = Math.floor(border.centerZ() - radius), maxZ = Math.ceil(border.centerZ() + radius);",
                   "        final double minZ = border.centerZ() - radius, maxZ = border.centerZ() + radius;"),
    "floor-z": (PPL,
                "        final double minZ = Math.floor(border.centerZ() - radius), maxZ = Math.ceil(border.centerZ() + radius);",
                "        final double minZ = border.centerZ() - radius, maxZ = Math.ceil(border.centerZ() + radius);"),
    "outside": (PPL,
                "        return beyond(to.x(), minX, maxX) > beyond(from.x(), minX, maxX)",
                "        if (!Boolean.getBoolean(\"proof.never\")) return false;\n"
                "        return beyond(to.x(), minX, maxX) > beyond(from.x(), minX, maxX)"),
}

path, old, new = VARIANTS[sys.argv[1]]
text = path.read_text()
assert text.count(old) == 1, (path.name, old)
line = text[:text.index(old)].count("\n") + 1
path.write_text(text.replace(old, new))
print(f"variant {sys.argv[1]}: {path.name}:{line}")

==================== summarize.py
"""Prints, for JUnit XML reports, the counts and every failure with its message and the test line it came from."""
import re
import sys
import xml.etree.ElementTree as ET

for path in sys.argv[1:]:
    root = ET.parse(path).getroot()
    name = root.get("name")
    print(f"{name}: tests={root.get('tests')} failures={root.get('failures')} errors={root.get('errors')} "
          f"skipped={root.get('skipped')} time={root.get('time')}")
    simple = name.rsplit(".", 1)[-1]
    for case in root.findall("testcase"):
        for kind in ("failure", "error"):
            node = case.find(kind)
            if node is None:
                continue
            message = (node.get("message") or "").splitlines()[0] if node.get("message") else ""
            text = node.text or ""
            line = re.search(re.escape(simple) + r"\.java:(\d+)", text)
            print(f"  {kind.upper()} {case.get('name')} at {simple}.java:{line.group(1) if line else '?'}: {message}")
