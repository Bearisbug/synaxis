#!/bin/zsh
# usage: g.sh <label> [nodb] -- <gradle args...>   runs one Gradle in the integrate worktree, one at a time
label=$1; shift
nodb=0
[ "$1" = "--" ] && shift
if [ "$1" = nodb ]; then nodb=1; shift; fi
source /tmp/m3e-run046/env.sh
if (( nodb )); then unset SYNAXIS_TEST_PG_URL SYNAXIS_TEST_PG_USER SYNAXIS_TEST_PG_PASSWORD SYNAXIS_TEST_REDIS_URL; fi
cd /Users/bug/Documents/Projects/Minecraft/synaxis-wt/m3e-integrate
while pgrep -f 'GradleWrapperMain|GradleDaemon|GradleMain' >/dev/null; do echo "waiting for another Gradle: $(pgrep -fl 'GradleWrapperMain|GradleDaemon|GradleMain' | cut -c1-80 | tr '\n' ';')"; sleep 20; done
log=/tmp/m3e-run046/logs/$label.log
start=$(date +%s)
echo "# $label $(date '+%F %T') HEAD $(git rev-parse --short=10 HEAD) load1 $(sysctl -n vm.loadavg | awk '{print $2}') swap_free $(sysctl -n vm.swapusage | sed -E 's/.*free = ([0-9.]+M).*/\1/') db=$((1-nodb))" > $log
echo "# ./gradlew $* --no-daemon --init-script gradle/init/aliyun-mirror.init.gradle.kts --max-workers=4" >> $log
./gradlew "$@" --no-daemon --init-script gradle/init/aliyun-mirror.init.gradle.kts --max-workers=4 >> $log 2>&1
rc=$?
echo "# rc=$rc wall=$(( $(date +%s) - start ))s end $(date '+%T')" >> $log
tail -1 $log; grep -E 'BUILD (SUCCESSFUL|FAILED)|tests completed' $log | head -3
exit $rc

=== xml.sh
#!/bin/zsh
setopt nullglob
# usage: xml.sh <newer-than-epoch> <class simple name>...   prints tests/skipped/failures and failure heads with test lines
since=$1; shift
cd /Users/bug/Documents/Projects/Minecraft/synaxis-wt/m3e-integrate
for c in "$@"; do
  for f in */build/test-results/test/TEST-*.$c.xml build/test-results/test/TEST-*.$c.xml; do
    m=$(stat -f %m $f); [ $m -lt $since ] && { echo "$c: stale $f"; continue; }
    head -2 $f | grep -o 'testsuite name="[^"]*" tests="[0-9]*" skipped="[0-9]*" failures="[0-9]*" errors="[0-9]*"' | sed 's/testsuite name="io.github.bearisbug.synaxis.//'
    perl -0ne 'while (/<testcase name="([^"]+)"[^>]*>\s*<failure message="([^"]{0,300})[^"]*"[^>]*>(.*?)<\/failure>/sg) { my ($n,$msg,$body)=($1,$2,$3); my @l = ($body =~ /at [\w.\$]+\((\w+Test\.java:\d+)\)/g); my %s; @l = grep { !$s{$_}++ } @l; $msg =~ s/&lt;/</g; $msg =~ s/&gt;/>/g; $msg =~ s/&quot;/"/g; $msg =~ s/&#10;/ /g; print "  FAIL $n @ ", join(",", @l[0..($#l<2?$#l:2)]), " : $msg\n"; }' $f
    perl -0ne 'if (/<skipped message="([^"]{0,200})/) { my $m=$1; $m =~ s/&quot;/"/g; print "  SKIP e.g. $m\n" }' $f
  done
done

=== mut.pl
#!/usr/bin/perl
# usage: mut.pl <file> <old> <new>   literal, the old text must occur exactly once
use strict; use warnings;
my ($file, $old, $new) = @ARGV;
my $root = "/Users/bug/Documents/Projects/Minecraft/synaxis-wt/m3e-integrate/";
$file = $root . $file unless $file =~ m{^/};
open(my $in, '<', $file) or die "open $file: $!"; local $/; my $text = <$in>; close $in;
my $count = () = $text =~ /\Q$old\E/g;
die "mut: '$old' occurs $count times in $file\n" unless $count == 1;
$text =~ s/\Q$old\E/$new/;
open(my $out, '>', $file) or die; print $out $text; close $out;
print "mut ok: $file\n";

#!/bin/zsh

run() { # run <label> <restore> <classes> -- gradle args
  local label=$1 restore=$2 classes=$3; shift 3; [ "$1" = "--" ] && shift
  local t0=$(date +%s)
  /tmp/m3e-run046/g.sh $label -- "$@" > /dev/null
  local rc=$?
  { echo "== $label rc=$rc $(tail -1 /tmp/m3e-run046/logs/$label.log)"; grep -E 'tests completed' /tmp/m3e-run046/logs/$label.log | head -2; /tmp/m3e-run046/xml.sh $t0 ${=classes}; } >> $OUT
  if [ -n "$restore" ]; then (cd $R && git checkout -- ${=restore}); fi
  (cd $R && git status --porcelain | grep -v '^ M docs/' | sed 's/^/  DIRTY /') >> $OUT
}
