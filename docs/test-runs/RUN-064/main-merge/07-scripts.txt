=== lock.zsh
# Sourced at the top level of a job script (not from a function: a zsh EXIT trap set inside a function fires when
# the function returns). Takes /tmp/synaxis-heavy.lock, retrying every 60 s; a lock whose owner pid is gone is stale.
LOCK=/tmp/synaxis-heavy.lock
until mkdir $LOCK 2>/dev/null; do
  o=$(cat $LOCK/owner 2>/dev/null); p=$(echo $o | sed -nE 's/.*pid=([0-9]+).*/\1/p')
  if [[ -n $p ]] && ! kill -0 $p 2>/dev/null; then log "stale lock (pid $p gone): $o; removing"; rm -rf $LOCK; continue; fi
  log "busy: $o"; sleep 60
done
trap 'rm -rf /tmp/synaxis-heavy.lock; echo "$(date "+%F %T") lock released"' EXIT
trap 'exit 143' TERM INT HUP
echo "owner=Synaxis光照/$STAGE pid=39601 since=$(date '+%FT%T%z') task=$TASK" > $LOCK/owner
log "lock taken: $(cat $LOCK/owner)"

# literal replacement, exactly one occurrence, fails loudly otherwise
subst() {
  local n=$(OLD="$2" perl -0ne 'my $o=$ENV{OLD}; my $c=()=/\Q$o\E/g; print $c' "$1")
  [[ $n == 1 ]] || { echo "  SUBST FAILED ($n matches) in $1: $2"; return 1; }
  OLD="$2" NEW="$3" perl -0pi -e 'my $o=$ENV{OLD}; my $n=$ENV{NEW}; s/\Q$o\E/$n/;' "$1"
  echo "  changed $1"
}
showdiff() { git diff -U0 -- "$@" | /usr/bin/grep -E '^[-+][^-+]' | sed 's/^/    diff /'; }
restore() { git checkout HEAD -- "$@"; echo "  restored; git status --short (outside docs): [$(git status --short | /usr/bin/grep -v ' docs/' | tr '\n' ' ')]"; }
# JUnit XML of one class: counts, failures with their messages, the test file lines in the traces
xmlsum() {
  local f=$1
  [[ -f $f ]] || { echo "  (no result file $f)"; return; }
  echo "  ${f:t}: $(/usr/bin/grep -o -m1 'tests="[0-9]*" skipped="[0-9]*" failures="[0-9]*" errors="[0-9]*"' $f) $(/usr/bin/grep -o -m1 'timestamp="[^"]*"' $f)"
  perl -0ne 'while (/<testcase name="([^"]*)"[^>]*>\s*<(failure|error) message="([^"]*)"[^>]*>(.*?)<\/\2>/sg) { my ($n,$m,$b)=($1,$3,$4); my @l = ($b =~ /\(([A-Za-z]+Test\.java:\d+)\)/g); my %s; @l = grep { !$s{$_}++ } @l; $m =~ s/&lt;/</g; $m =~ s/&gt;/>/g; $m =~ s/&quot;/"/g; $m =~ s/&#39;/\x27/g; $m =~ s/&amp;/&/g; $m = substr($m,0,600); print "    FAIL $n: $m\n      at @l\n"; }' $f
  perl -0ne 'while (/<testcase name="([^"]*)"[^>]*>\s*<skipped/sg) { print "    SKIPPED $1\n"; }' $f
}
